```typescript
// 主要功能：
// 1. getAvailableTraits() - 获取可用特征列表
// 2. useTraitOnCustomer(traitId) - 使用特征压价
// 3. calculatePriceReduction(trait) - 计算压价幅度
// 4. markTraitAsUsed(traitId) - 标记特征已使用

export function useTraitLeverage(item: Item, customer: Customer) {
  const availableTraits = useMemo(() =>
    item.revealedTraits.filter(t =>
      !item.traitUsage.find(u => u.traitId === t.id && u.isUsed)
    ), [item]);

  const useTraitOnCustomer = (traitId: string) => {
    const trait = item.revealedTraits.find(t => t.id === traitId);
    if (!trait) return;

    // 根据特征类型计算影响
    const priceReduction = calculatePriceReduction(trait);
    const patienceCost = trait.type === 'FLAW' ? 0 : 1;

    // dispatch action
    dispatch({
      type: 'USE_TRAIT_LEVERAGE',
      payload: { traitId, priceReduction, patienceCost }
    });
  };

  return { availableTraits, useTraitOnCustomer };
}
```

```typescript
interface MailContext {
  itemName?: string;
  amount?: number;
  playerName?: string;
  npcName?: string;
  daysPassed?: number;
  interestPaid?: number;
}

export function interpolateMail(
  template: string,
  context: MailContext
): string {
  return template
    .replace(/\{\{itemName\}\}/g, context.itemName || '物品')
    .replace(/\{\{amount\}\}/g, formatCurrency(context.amount || 0))
    .replace(/\{\{playerName\}\}/g, context.playerName || '老板')
    .replace(/\{\{npcName\}\}/g, context.npcName || '顾客')
    .replace(/\{\{daysPassed\}\}/g, String(context.daysPassed || 0))
    .replace(/\{\{interestPaid\}\}/g, formatCurrency(context.interestPaid || 0));
}

function formatCurrency(amount: number): string {
  return `$${amount.toLocaleString()}`;
}
```

#### 4.2 创建检查函数

**文件**: `systems/narrative/reputationEvents.ts`

```typescript
export function checkReputationThresholds(
  reputation: ReputationProfile,
  triggeredEvents: string[] // 已触发的一次性事件
): ReputationEventEffect[] {
  const effects: ReputationEventEffect[] = [];

  for (const event of REPUTATION_EVENTS) {
    // 跳过已触发的一次性事件
    if (event.triggerOnce && triggeredEvents.includes(event.id)) {
      continue;
    }

    const value = reputation[event.axis];
    const triggered = event.comparison === 'above'
      ? value >= event.threshold
      : value <= event.threshold;

    if (triggered) {
      effects.push(event.effect);
      if (event.triggerOnce) {
        triggeredEvents.push(event.id);
      }
    }
  }

  return effects;
}
```

```typescript
// 在 findEligibleEvent 中添加续当请求检查
function checkRenewalRequests(state: GameState): RenewalRequestEvent | null {
  const soonToExpire = state.inventory.filter(item =>
    item.status === 'ACTIVE' &&
    item.pawnInfo.dueDate - state.day <= 2 && // 2天内到期
    item.relatedChainId // 有关联故事链
  );

  for (const item of soonToExpire) {
    const chain = state.activeChains.find(c => c.id === item.relatedChainId);
    if (chain && shouldRequestRenewal(chain)) {
      return generateRenewalRequest(item, chain);
    }
  }
  return null;
}

function shouldRequestRenewal(chain: EventChainState): boolean {
  // 基于链变量判断是否会来请求续当
  const funds = chain.variables.funds || 0;
  const hope = chain.variables.hope || 50;
  return funds < 500 && hope > 20; // 缺钱但还有希望
}
```

```typescript
export function generateFateHint(variables: ChainVariables): string | null {
  const applicableHints: string[] = [];

  for (const hint of FATE_HINTS) {
    const value = variables[hint.variableName];
    if (value !== undefined && hint.condition(value)) {
      applicableHints.push(...hint.hints);
    }
  }

  if (applicableHints.length === 0) return null;
  return applicableHints[Math.floor(Math.random() * applicableHints.length)];
}
```

```typescript
function MotherVisitPanel() {
  const { motherCondition, visitMother, hasVisitedToday } = useMotherHealth();
  const [currentDialogue, setCurrentDialogue] = useState<MotherDialogue | null>(null);
  const [lineIndex, setLineIndex] = useState(0);

  useEffect(() => {
    const dialogue = selectDialogue(motherCondition);
    setCurrentDialogue(dialogue);
  }, []);

  const handleContinue = () => {
    if (lineIndex < currentDialogue.lines.length - 1) {
      setLineIndex(prev => prev + 1);
    } else {
      visitMother(); // 完成探望
    }
  };

  if (hasVisitedToday) {
    return <div>今天已经探望过了。</div>;
  }

  const currentLine = currentDialogue?.lines[lineIndex];

  return (
    <div className="mother-visit-panel">
      {/* 病房背景 */}
      <div className="hospital-backdrop" />

      {/* 母亲立绘 */}
      <div className="mother-portrait">
        <img
          src="/assets/mother.png"
          className={motherCondition.status === 'Critical' ? 'grayscale' : ''}
        />
      </div>

      {/* 对话框 */}
      <div className={`dialogue-box ${currentLine?.speaker}`}>
        <span className="speaker-name">
          {currentLine?.speaker === 'mother' ? '母亲' : '我'}
        </span>
        <TypewriterText text={currentLine?.text} />
      </div>

      {/* 继续按钮 */}
      <button onClick={handleContinue} className="continue-btn">
        {lineIndex < currentDialogue.lines.length - 1 ? '继续' : '离开病房'}
      </button>
    </div>
  );
}
```
